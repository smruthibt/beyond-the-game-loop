cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(Towerfall LANGUAGES CXX)
    set(BUILD_TOWERFALL OFF CACHE BOOL "" FORCE)
    add_subdirectory(.. engine)
    enable_testing()
endif()
add_library(towerfall_rules src/Simulation.cpp src/Match.cpp)
target_include_directories(towerfall_rules PUBLIC src)
target_link_libraries(towerfall_rules PUBLIC engine)
add_executable(towerfall src/App.cpp src/Transport.cpp src/Presentation.cpp)
target_link_libraries(towerfall PRIVATE towerfall_rules)
add_executable(towerfall_tests tests/RulesTests.cpp)
target_link_libraries(towerfall_tests PRIVATE towerfall_rules)
if(NOT MSVC)
    target_compile_options(towerfall_rules PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(towerfall PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME towerfall_rules COMMAND towerfall_tests)
